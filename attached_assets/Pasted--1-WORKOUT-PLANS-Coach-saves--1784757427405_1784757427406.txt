-- ══════════════════════════════════════════════════════════
-- 1. WORKOUT PLANS
--    Coach saves the exercise program + cardio protocol here
-- ══════════════════════════════════════════════════════════
create table if not exists public.workout_plans (
  id         uuid primary key default gen_random_uuid(),
  client_id  uuid not null,
  coach_id   uuid not null,
  workouts   jsonb not null default '{}',   -- exercises, principles, cardioWeekStart
  cardio     jsonb not null default '[]',   -- array of cardio protocol objects
  updated_at timestamptz default now(),
  created_at timestamptz default now()
);

-- Open RLS so the anon key can read/write (matches your fake-auth setup)
alter table public.workout_plans enable row level security;

drop policy if exists "open" on public.workout_plans;
create policy "open" on public.workout_plans
  for all using (true) with check (true);


-- ══════════════════════════════════════════════════════════
-- 2. CLIENT WORKOUT LOGS
--    Client saves weekly weight/reps + cardio log here
-- ══════════════════════════════════════════════════════════
create table if not exists public.client_workout_logs (
  id         uuid primary key default gen_random_uuid(),
  client_id  uuid not null,
  week       integer not null,
  logs       jsonb not null default '{}',   -- keyed by `{week}_{exId}_{setIdx}` and `{week}_cardio_{Day}_*`
  saved_at   timestamptz default now(),
  unique (client_id, week)                  -- required for upsert (merge-duplicates)
);

alter table public.client_workout_logs enable row level security;

drop policy if exists "open" on public.client_workout_logs;
create policy "open" on public.client_workout_logs
  for all using (true) with check (true);

-- ══════════════════════════════════════════════════════════
-- 4. LAB COMMENTS  (coach ↔ client discussion on a lab)
-- ══════════════════════════════════════════════════════════
create table if not exists public.lab_comments (
  id          uuid primary key default gen_random_uuid(),
